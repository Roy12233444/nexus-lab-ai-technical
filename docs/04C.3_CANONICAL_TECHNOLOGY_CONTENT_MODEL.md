# 3. 🏗️ TypeScript Type Schemas (`src/types/technology.ts`)

```typescript
export type TechnologyDomainId =
  | 'reasoning'
  | 'agent_execution'
  | 'memory'
  | 'fault_tolerance'
  | 'safety_governance'
  | 'distributed_space';

export type TechnologySubstrateCategory =
  'NEXUS_INVENTION' | 'SUPPORTING_STACK' | 'EXTERNAL_SERVICE';

export type SpecificationState = 'SPECIFIED' | 'UNSPECIFIED';

export interface TechnologyState {
  maturity: MaturityState;
  specification: SpecificationState;
  evidence: EvidenceState;
  temporal: TemporalState;
  publicRelevance: PublicRelevanceState;

export interface TechnologyDomain {
  id: TechnologyDomainId;
  name: string;
  tagline: string;
  description: string;
  coreChallenges: string[];
  capabilityIds: string[];
}

export interface TechnologyCapability {
  id: string;
  domainId: TechnologyDomainId;
  name: string;
  description: string;
  technologyIds: string[];
  projectIds: string[];
}

export interface TechnologySubstrate {
  id: string;
  domainId: TechnologyDomainId;
  name: string;
  category: TechnologySubstrateCategory;
  description: string;
  state: TechnologyState;
  claimIds: string[];
}

export interface TechnologyClaim {
  id: string;
  technologyId: string;
  statement: string;
  sourceDoc: string;
  authorityTier: AuthorityTier;

}

export interface QuarantinedTechnologyClaim {
  id: string;

  quarantineReason: string;
  requiredVerification: string;
  publicAllowed: false;
}
```

# 5. 🛡️ Semantic Invariant Rules (`RULE-001` through `RULE-008`)

|   Rule ID    | Invariant Title            | Description                                                                                              | Enforcing Function            |
| :----------: | :------------------------- | :------------------------------------------------------------------------------------------------------- | :---------------------------- |
| **RULE-001** | Provenance Requirement     | Every substrate must link to valid claims with source documents and authority tiers.                     | `validateTechnologyContent()` |
| **RULE-002** | Quarantine Firewall        | Quarantined claims must enforce `publicAllowed: false` and cannot enter public presentation selectors.   | `validateTechnologyContent()` |
| **RULE-003** | Maturity Decoupling        | Maturity, Evidence, and Temporal states must remain distinct enums.                                      | `validateTechnologyContent()` |
| **RULE-004** | Project Independence       | No technology capability may create unevidenced operational coupling between sovereign projects.         | `validateTechnologyContent()` |
| **RULE-005** | Supporting Tech Separation | Open-source dependencies (Rust, Python, Docker) cannot be labeled `NEXUS_INVENTION`.                     | `validateTechnologyContent()` |
| **RULE-006** | Exclusion Enforcement      | Claims marked `EXCLUDED` or `UNKNOWN` (e.g. `TECH-016`) are blocked from public presentation.            | `validateTechnologyContent()` |
| **RULE-007** | Temporal Consistency       | `FUTURE` substrates cannot be marked `CURRENT` or `IMPLEMENTED`.                                         | `validateTechnologyContent()` |
| **RULE-008** | Referential Integrity      | No orphaned references across `domainId`, `capabilityIds`, `technologyIds`, `projectIds`, or `claimIds`. | `validateTechnologyContent()` |

# 7. 🔒 Downstream Consumption Contract

All downstream tasks in the Technology Track must adhere to the following contract:

1. **04C.4 (Page Architecture):** Must structure the Technology page narrative directly around the 6 canonical domains and the Three-Tier Capability Hierarchy.
2. **04C.5 (UI Implementation):** Must consume `CANONICAL_TECHNOLOGY_REGISTRY` via programmatic selectors (`getPublicTechnologyDomains()`, `getPublicTechnologyCapabilities()`, `getPublicTechnologySubstrates()`, `getPublicTechnologyClaims()`). No hardcoded claims or invented statistics in React components.
3. **04C.6 (Verification & Lock):** Must run `validateTechnologyContent()` in the lock verification test suite to guarantee zero data regression.
